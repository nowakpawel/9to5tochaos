cmake_minimum_required(VERSION 3.16)
project(9to5chaos VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
find_package(glm REQUIRED)
file(GLOB_RECURSE SOURCES "src/*.cpp")
add_executable(9to5chaos ${SOURCES})
target_include_directories(9to5chaos PRIVATE src)
target_link_libraries(9to5chaos SDL2::SDL2 OpenGL::GL GLEW::GLEW glm::glm)
target_compile_options(9to5chaos PRIVATE -Wall -Wextra -O2)
set_target_properties(9to5chaos PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
