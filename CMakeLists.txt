cmake_minimum_required(VERSION 3.16)
project(9to5chaos VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)
find_package(glm REQUIRED)

# Use pkg-config for GLEW (handles non-standard lib paths like /usr/lib64)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLEW REQUIRED glew)

file(GLOB_RECURSE SOURCES "src/*.cpp")
add_executable(9to5chaos ${SOURCES})
target_include_directories(9to5chaos PRIVATE src ${GLEW_INCLUDE_DIRS})
target_link_libraries(9to5chaos SDL2::SDL2 OpenGL::GL glm::glm ${GLEW_LIBRARIES})
target_link_directories(9to5chaos PRIVATE ${GLEW_LIBRARY_DIRS})
target_compile_options(9to5chaos PRIVATE -Wall -Wextra -O2)
set_target_properties(9to5chaos PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
